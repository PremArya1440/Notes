============eksctl install
1)ARCH=amd64
2)PLATFORM=$(uname -s)_$ARCH
3)curl -sLO "https://github.com/eksctl-io/eksctl/releases/latest/download/eksctl_$PLATFORM.tar.gz"

4)tar -xzf eksctl_$PLATFORM.tar.gz -C /tmp && rm eksctl_$PLATFORM.tar.gz
5)sudo install -m 0755 /tmp/eksctl /usr/local/bin && rm /tmp/eksctl
6)eksctl version
